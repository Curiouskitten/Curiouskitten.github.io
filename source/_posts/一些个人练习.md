## 时间计算
``` bash
from datetime import datetime
import time


def calculate_time_diff(start_time, end_time):
    # 将时间字符串转换成datetime对象
    start = datetime.strptime(start_time, '%Y-%m-%d %H:%M:%S')
    end = datetime.strptime(end_time, '%Y-%m-%d %H:%M:%S')
    # 计算时间差值
    time_difference = end - start
    # 将时间差值换算成小时
    hours = time_difference.total_seconds() / 3600
    return hours


def get_curt():
    """
    :return:返回当前公历时间%Y-%m-%d %H:%M:%S
    """
    current_time = datetime.now().strftime('%Y-%m-%d %H:%M:%S')
    return current_time


def get_timestamp(time_sample=None):
    """
    将指定时间转换成时间戳，毫秒级
    :param time_sample: 输入的公历时间，默认为None获取当前时间
    :return: 13位毫秒级时间戳
    """
    if time_sample is None:
        time_stamp = time.time_ns() // 10**6
    elif time_sample == '':
        time_stamp = time.time_ns() // 10**6
    else:
        dt_obj = datetime.strptime(time_sample, '%Y-%m-%d %H:%M:%S')
        time_stamp = round(dt_obj.timestamp() * 1000)
    return time_stamp


def stamp_to_time(stamp_time=None):
    """
    将时间戳（毫秒级）转换成公历格式时间
    :param stamp_time:
    :return:
    """
    if stamp_time is None:
        date_time = get_curt()
    elif stamp_time == '':
        date_time = get_curt()
    else:
        stamp_time_sec = stamp_time / 1000
        date_time = datetime.fromtimestamp(stamp_time_sec)
    return date_time


if __name__ == '__main__':
    workCompleted_time = '2024-03-27 17:24:38'
    permission_time = '2024-03-27 17:24:18'
    planEndTime = '2024-03-29 18:00:00'
    planStartTime = '2024-03-29 08:00:00'

    plan_diff = calculate_time_diff(planEndTime, planStartTime)
    act_diff = calculate_time_diff(workCompleted_time, permission_time)
    all_diff = act_diff - plan_diff
    print('计划工时差值{}'.format(abs(plan_diff)))
    print('实际工时差值{}'.format(abs(act_diff)))
    print('两者差值{}'.format(abs(all_diff)))
    time_res = get_curt()
    print(time_res)
    ts = get_timestamp()
    print(ts)
    date_time = stamp_to_time(ts)
    print(date_time)
```
